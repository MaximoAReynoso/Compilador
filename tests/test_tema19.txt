INTEGER a, b, c;

INTEGER FUNCTION f(INTEGER p1, INTEGER p2, INTEGER p3)
BEGIN
    ret(p1 + p2 + p3);
END;

BEGIN
    a := 1$i;
    b := 2$i;

    // Caso 1: Orden [1$i, 2$i, 3$i]
    c := f(a, b = (4$i), a + b)[1$i, 2$i, 3$i];
    pout(c);

    // Restaurar a y b
    a := 1$i;
    b := 2$i;

    // Caso 2: Orden [1$i, 3$i, 2$i] (evalua a + b antes de b = 4)
    c := f(a, b = (4$i), a + b)[1$i, 3$i, 2$i];
    pout(c);

    pout("fin");
END;
